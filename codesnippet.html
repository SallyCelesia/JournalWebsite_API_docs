POST /auth/register

Handles user registration by hashing password with bcrypt and creating a user record.

Express Route Code
app.post('/auth/register', async (req, res) => {
  try {
    const { name, email, password } = req.body;
    if (!name || !email || !password) {
      return res.status(400).json({ error: 'name, email and password are required' });
    }
    if (typeof password !== 'string' || password.length < 8) {
      return res.status(400).json({ error: 'password must be at least 8 characters' });
    }
    const existing = await User.findOne({ email: email.toLowerCase().trim() });
    if (existing) {
      return res.status(409).json({ error: 'An account with this email already exists' });
    }
    const hash = await bcrypt.hash(password, 10);
    const user = await User.create({ name, email, password: hash });
    res.status(201).json(publicUser(user));
  } catch (err) {
    sendError(res, err);
  }
});

POST /auth/login

Verifies email and password hash, then returns a signed 1-hour JWT token with user details.

Express Route Code
app.post('/auth/login', async (req, res) => {
  try {
    const { email, password } = req.body;
    if (!email || !password) {
      return res.status(400).json({ error: 'email and password are required' });
    }
    const user = await User.findOne({ email: email.toLowerCase().trim() }).select('+password');
    if (!user) {
      return res.status(401).json({ error: 'Invalid email or password' });
    }
    const matches = await bcrypt.compare(password, user.password);
    if (!matches) {
      return res.status(401).json({ error: 'Invalid email or password' });
    }

    const token = jwt.sign({ sub: user._id.toString() }, process.env.JWT_SECRET, { expiresIn: '1h' });
    res.json({ token, user: publicUser(user) });
  } catch (err) {
    sendError(res, err);
  }
});

PATCH /users/:id

Updates user name or email with validation and handles potential duplicate email errors.

Express Route Code
app.patch('/users/:id', requireLogin, requireOwner, async (req, res) => {
  try {
    if (!mongoose.isValidObjectId(req.params.id)) {
      return res.status(400).json({ error: 'Invalid id' });
    }
    const updates = {};
    if (req.body.name !== undefined) updates.name = req.body.name;
    if (req.body.email !== undefined) updates.email = req.body.email;

    if (Object.keys(updates).length === 0) {
      return res.status(400).json({ error: 'Send at least one of name or email' });
    }

    const user = await User.findByIdAndUpdate(
      req.params.id,
      updates,
      { new: true, runValidators: true }
    );
    if (!user) {
      return res.status(404).json({ error: 'User not found' });
    }
    res.json(publicUser(user));
  } catch (err) {
    if (err.code === 11000) {
      return res.status(409).json({ error: 'That email is already in use' });
    }
    sendError(res, err);
  }
});

GET /:userId/:month

Queries journal entries for a given user and month (1-12), ordered chronologically by day.

Express Route Code
app.get('/:userId/:month', requireLogin, requireOwner, async (req, res) => {
  try {
    if (!mongoose.isValidObjectId(req.params.userId)) {
      return res.status(400).json({ error: 'Invalid user id' });
    }
    const month = parseWholeNumber(req.params.month);
    if (month === null || month < 1 || month > 12) {
      return res.status(400).json({ error: 'month must be a number from 1 to 12' });
    }
    const entries = await Journal.find({ user: req.params.userId, month }).sort({ day: 1 });
    res.json(entries);
  } catch (err) {
    sendError(res, err);
  }
});

POST /:userId/:month

Creates a journal entry for the specified day. Rejects duplicates using MongoDB unique compound index.

Express Route Code
app.post('/:userId/:month', requireLogin, requireOwner, async (req, res) => {
  try {
    if (!mongoose.isValidObjectId(req.params.userId)) {
      return res.status(400).json({ error: 'Invalid user id' });
    }
    const month = parseWholeNumber(req.params.month);
    if (month === null || month < 1 || month > 12) {
      return res.status(400).json({ error: 'month must be a number from 1 to 12' });
    }
    const { day, content } = req.body;
    if (day === undefined || content === undefined) {
      return res.status(400).json({ error: 'day and content are required' });
    }
    const user = await User.findById(req.params.userId);
    if (!user) {
      return res.status(404).json({ error: 'User not found' });
    }
    const entry = await Journal.create({ user: user._id, month, day, content });
    res.status(201).json(entry);
  } catch (err) {
    if (err.code === 11000) {
      return res.status(409).json({ error: 'This user already has an entry for that day' });
    }
    sendError(res, err);
  }
});
